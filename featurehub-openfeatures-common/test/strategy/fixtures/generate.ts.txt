// Regenerates typescript_golden.json from the TypeScript SDK
// (featurehub-javascript-sdk/packages/core):
//
//   1. replace FEATUREHUB_JS_CORE below with the path to packages/core
//   2. NODE_PATH=<packages/core>/node_modules esbuild generate.ts --bundle \
//        --platform=node --format=esm --outfile=generate.mjs
//   3. TZ=UTC node generate.mjs typescript_golden.json
//
// Generates golden strategy-evaluation results from the TypeScript SDK, so the
// Dart port can be checked against the real implementation.
//
// One deliberate difference: NUMBER attributes use strict numbers
// (rollout-strategies.yaml js_conversions.strict_number), not JavaScript's
// parseInt / parseFloat. Where TypeScript's result differs, the case records
// the strict result as `result` and TypeScript's as `typescript`.

// rollout-strategies.yaml js_conversions.strict_number: a finite number, or a
// string that is exactly an optional sign, digits and an optional fraction
function strictNumber(v: any): number | undefined {
  if (typeof v === "number") return Number.isFinite(v) ? v : undefined;
  return typeof v === "string" && /^[+-]?[0-9]+(\.[0-9]+)?$/.test(v) ? Number(v) : undefined;
}

// matchers.NUMBER with strict numbers: a supplied value that is not a number
// never matches; values that are not numbers are ignored
function strictNumberMatch(conditional: string, values: any[], supplied: any): boolean {
  const n = strictNumber(supplied);
  if (n === undefined) return false;
  const valid = values.filter((v) => v != null && strictNumber(v) !== undefined);
  const nums = valid.map((v) => strictNumber(v)!);
  const str = String(n);
  const strs = valid.map((v) => String(v));
  const regex = (p: string) => { try { return new RegExp(p).test(str); } catch (_) { return false; } };
  switch (conditional) {
    case C.Equals: return nums.some((v) => v === n);
    case C.NotEquals: return !nums.some((v) => v === n);
    case C.Greater: return nums.some((v) => n > v);
    case C.GreaterEquals: return nums.some((v) => n >= v);
    case C.Less: return nums.some((v) => n < v);
    case C.LessEquals: return nums.some((v) => n <= v);
    case C.Includes: return strs.some((v) => str.includes(v));
    case C.Excludes: return !strs.some((v) => str.includes(v));
    case C.Regex: return strs.some(regex);
    case C.EndsWith: return strs.some((v) => str.endsWith(v));
    case C.StartsWith: return strs.some((v) => str.startsWith(v));
  }
  throw new Error(`unknown conditional ${conditional}`);
}

function matchers(type: string, valueSets: any[][], supplied: any[]) {
  for (const conditional of conds) {
    for (const values of valueSets) {
      for (const s of supplied) {
        const attr: any = { conditional, type, values, fieldName: "f" };
        let result: any;
        try {
          result = reg.findMatcher(attr).match(s, attr);
          if (result === undefined) result = "undefined";
        } catch (_) {
          result = "throw";
        }
        if (type === T.Number) {
          const strict = strictNumberMatch(conditional, values, s);
          if (strict !== result) {
            out.matchers.push({ type, conditional, values, supplied: s, result: strict, typescript: result });
            continue;
          }
        }
        out.matchers.push({ type, conditional, values, supplied: s, result });
      }
    }
  }
}

// ---- apply
const af = new ApplyFeature();
function ctx(attrs: Record<string, any>) {
  return {
    getAttr: (k: string, d?: any) => (k in attrs ? attrs[k] : d),
    defaultPercentageKey: () =>
      "session" in attrs ? attrs["session"]?.toString() : attrs["userkey"],
  } as any;
}
function apply(name: string, strategies: any[], featureValueId: string, contexts: any[]) {
  for (const c of contexts) {
    const r = af.apply(strategies, "key", featureValueId, ctx(c));
    out.apply.push({
      name, strategies, featureValueId, context: c,
      matched: r.matched, value: r.value ?? null, strategyId: r.strategyId ?? null,
    });
  }
}
